registerUser(displayName, email, password, role)
	IF email NOT valid OR password length < 8 THEN 
	  SHOW error "Invalid input"
    	  RETURN
	END IF
     TRY
	CREATE user in FireBase 
Auth with email, password
	GET uid from Auth
	CREATE User Object:
  	    displayName = displayName
 	    email = email 
	    role = role // learner or admin
	 createdAt = current timestamp

	SAVE to FireBase REST API:
	 POST /user/{uid} with User Object

	SHOW "Registered successfully"
	REDIRECT to dashboard.html
     CATCH error.message
END FUNCITION

FUNCTION checkAuthState()
  ON Auth state changed
    IF user exists THEN 
       FETCH /user/{uid} from dataBase 
       IF user.role == "admin" THEN
         REDIRECT to admin.html 
       ELSE
         REDIRECT to dashboard.html // learner.html
       END IF
       ELSE
         REDIRECT to login.html
       END IF
END FUNCTION

FUNCTION laodDashBoard(userId)
  FETCH all task: GET /tasks
  FILTER task WHERE task.userId == currentUser.uid // array filter
  total = task.length
  completedList = FILTER task WHERE completed == true
  completed = completedList.length
  outstanding = total - completed
  progress = (completed / total) * 100
  
  DISPLAY on Dashboard:
   taskTotals = { total, completed, outstanding }
   progressBar = progress

   taskTitles = MAP tasks to titles
   highPriority = FILTER task WHERE priority == "high"

   FOR EACH task IN tasks
    CREATE DOM element dynamically
    APPEND to task container
   END FOR 
END FUNCTION
